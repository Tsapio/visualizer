            async initAudio() {

                if (!this.audioCtx) {

                    this.audioCtx = new (window.AudioContext || window.webkitAudioContext)();

                }

                if (this.audioCtx.state === 'suspended') {

                    await this.audioCtx.resume();

                }

            }

            setupStream(stream, connectToOutput) {

                if (this.stream) this.stop();

                this.stream = stream;

                this.source = this.audioCtx.createMediaStreamSource(stream);

                this.analyser = this.audioCtx.createAnalyser();

                this.gainNode = this.audioCtx.createGain();

                this.analyser.fftSize = 2048;

                this.analyser.smoothingTimeConstant = 0.85;

                this.gainNode.gain.value = this.gain;

                this.source.connect(this.gainNode);

                this.gainNode.connect(this.analyser);

                if (connectToOutput) this.gainNode.connect(this.audioCtx.destination);

                stream.getTracks().forEach(track => { track.onended = () => this.stop(); });

                this.isPlaying = true;

                this.isPaused = false;

                ui.showControls();

                this.loop();

            }

            loop() {

                if (!this.isPlaying || this.isPaused) return;

                this.rafId = requestAnimationFrame(() => this.loop());

                this.ctx.fillStyle = 'rgba(15, 23, 42, 0.25)';

                this.ctx.fillRect(0, 0, this.width, this.height);

                const bufferLength = this.analyser.frequencyBinCount;

                const dataArray = new Uint8Array(bufferLength);

                let rms = 0;


                if (this.mode === 'waveform') {

                    this.analyser.getByteTimeDomainData(dataArray);

                    this.drawWaveform(dataArray, bufferLength);

                    for(let i=0; i<bufferLength; i++) {

                        let val = (dataArray[i] - 128) / 128.0;

                        rms += val * val;

                    }

                } else {

                    this.analyser.getByteFrequencyData(dataArray);

                    if (this.mode === 'frequency') this.drawFrequency(dataArray, bufferLength);

                    if (this.mode === 'circular') this.drawCircular(dataArray, bufferLength);

                    for(let i=0; i<bufferLength; i++) {

                        rms += (dataArray[i]/255) * (dataArray[i]/255);

                    }

                }


                if (Math.random() > 0.8) {

                    rms = Math.sqrt(rms / bufferLength);

                    const db = 20 * Math.log10(rms);

                    const displayDb = Math.max(-100, Math.round(db));

                    this.hudDb.textContent = isFinite(db) ? displayDb : '-∞';

                    const pct = Math.min(100, Math.max(0, (displayDb + 60) * 2)); 

                    this.hudBar.style.width = `${pct}%`;

                }

            }


            drawWaveform(data, bufferLen) {

                this.ctx.lineWidth = 2.5;

                this.ctx.strokeStyle = this.theme.primary;

                this.ctx.shadowColor = this.theme.secondary;

                this.ctx.shadowBlur = 10;

                this.ctx.beginPath();

                const sliceWidth = this.width / bufferLen;

                let x = 0;

                for (let i = 0; i < bufferLen; i++) {

                    const v = data[i] / 128.0;

                    const y = v * (this.height / 2);

                    if (i === 0) this.ctx.moveTo(x, y);

                    else this.ctx.lineTo(x, y);

                    x += sliceWidth;

                }

                this.ctx.lineTo(this.width, this.height/2);

                this.ctx.stroke();

                this.ctx.shadowBlur = 0;

            }


            drawFrequency(data, bufferLen) {

                const barWidth = (this.width / bufferLen) * 2.5;

                let x = 0;

                for(let i = 0; i < bufferLen; i++) {

                    const barHeight = (data[i] / 255) * this.height;

                    const grad = this.ctx.createLinearGradient(0, this.height, 0, this.height - barHeight);

                    grad.addColorStop(0, this.theme.secondary);

                    grad.addColorStop(1, this.theme.primary);

                    this.ctx.fillStyle = grad;

                    this.ctx.fillRect(x, this.height - barHeight, barWidth, barHeight);

                    x += barWidth + 1;

                    if(x > this.width) break;

                }

            }


            drawCircular(data, bufferLen) {

                const cx = this.width / 2;

                const cy = this.height / 2;

                const r = Math.min(cx, cy) * 0.4;

                this.ctx.lineWidth = 2;

                this.ctx.strokeStyle = this.theme.primary;

                this.ctx.shadowColor = this.theme.secondary;

                this.ctx.shadowBlur = 15;

                this.ctx.beginPath();

                const len = bufferLen / 2;

                for (let i = 0; i < len; i++) {

                    const v = data[i] / 255.0;

                    const angle = (i / len) * Math.PI * 2;

                    const amp = r + (v * 100 * this.gain);

                    const x = cx + Math.cos(angle) * amp;

                    const y = cy + Math.sin(angle) * amp;

                    if(i===0) this.ctx.moveTo(x,y);

                    else this.ctx.lineTo(x,y);

                }

                this.ctx.closePath();

                this.ctx.stroke();

                this.ctx.beginPath();

                this.ctx.arc(cx, cy, r * 0.8, 0, Math.PI * 2);

                this.ctx.fillStyle = this.theme.secondary + '20'; 

                this.ctx.fill();

                this.ctx.shadowBlur = 0;

            }

        }
